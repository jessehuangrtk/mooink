cmake_minimum_required(VERSION 3.24)
set(PICO_BOARD pico2_w CACHE STRING "Board")
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake" OR NOT EXISTS "${FREERTOS_KERNEL_PATH}/portable/ThirdParty/GCC/RP2350_ARM_NTZ/CMakeLists.txt" OR NOT EXISTS "${LIBSMB2_PATH}/CMakeLists.txt")
  message(FATAL_ERROR "Run build_smb_wsl.sh to obtain pinned dependencies")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(pico_wifi_display_smb_probe C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
set(FREERTOS_CONFIG_FILE_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}")
add_subdirectory("${FREERTOS_KERNEL_PATH}/portable/ThirdParty/GCC/RP2350_ARM_NTZ" freertos)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ENABLE_UTILS OFF CACHE BOOL "" FORCE)
set(ENABLE_LIBKRB5 OFF CACHE BOOL "" FORCE)
set(ENABLE_GSSAPI OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIBSMB2_PATH}" libsmb2)
target_compile_definitions(libsmb2 PRIVATE HAVE_CONFIG_H PICO_PLATFORM=1 NO_SYS=0)
target_include_directories(libsmb2 BEFORE PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${LIBSMB2_PATH}/include/picow")
# Upstream Pico W port expects lwIP socket headers, FreeRTOS and SDK config.
target_link_libraries(libsmb2 PRIVATE pico_cyw43_arch_lwip_sys_freertos FreeRTOS-Kernel)
add_executable(pico_wifi_display_smb_probe ../smb_main.c ../smb_snapshot.c)
target_include_directories(pico_wifi_display_smb_probe PRIVATE
    "${CMAKE_CURRENT_LIST_DIR}" "${CMAKE_CURRENT_LIST_DIR}/.." "${LIBSMB2_PATH}/include" "${LIBSMB2_PATH}/include/picow")
target_compile_definitions(pico_wifi_display_smb_probe PRIVATE PICO_PLATFORM=1 NO_SYS=0)
target_link_libraries(pico_wifi_display_smb_probe PRIVATE pico_stdlib pico_sha256
    pico_cyw43_arch_lwip_sys_freertos FreeRTOS-Kernel-Heap4 libsmb2)
pico_enable_stdio_usb(pico_wifi_display_smb_probe 1)
pico_enable_stdio_uart(pico_wifi_display_smb_probe 1)
pico_add_extra_outputs(pico_wifi_display_smb_probe)
