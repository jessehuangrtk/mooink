cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2_w CACHE STRING "Board")
if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(pico2w_usb2bluetooth C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(pico2w_usb2bluetooth main.c)
target_include_directories(pico2w_usb2bluetooth PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(pico2w_usb2bluetooth PRIVATE pico_stdlib pico_cyw43_arch_none pico_btstack_cyw43 pico_btstack_classic pico_btstack_ble)
pico_enable_stdio_usb(pico2w_usb2bluetooth 1)
pico_enable_stdio_uart(pico2w_usb2bluetooth 0)
pico_add_extra_outputs(pico2w_usb2bluetooth)
